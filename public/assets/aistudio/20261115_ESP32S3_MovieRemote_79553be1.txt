void setLed(uint8_t r, uint8_t g, uint8_t b) {

// --- WEBSERVER HANDLER ---
void handleSkipAd() {
  String screenName = defaultScreen;

  // Controleer of een specifiek scherm is opgevraagd (?screen=lg / rog / custom)
  if (server.hasArg("screen")) {
    String reqScreen = server.arg("screen");
    reqScreen.toLowerCase();
    if (reqScreen == "lg" || reqScreen == "rog" || reqScreen == "custom") {
      screenName = reqScreen;
    }
  }

  // Haal de juiste opgeslagen waarden op
  int moveX = lgX;
  int moveY = lgY;
  if (screenName == "rog") {
    moveX = rogX;
    moveY = rogY;
  } else if (screenName == "custom") {
    moveX = customX;
    moveY = customY;
  }

  String activeMode = "Preset: " + screenName;

  // Overschrijf tijdelijk als er test-coördinaten (?x=...&y=...) zijn meegegeven
  bool hasCustomCoords = (server.hasArg("x") && server.hasArg("y"));
  if (hasCustomCoords) {
    moveX = server.arg("x").toInt();
    moveY = server.arg("y").toInt();
    activeMode = "Test (" + screenName + ")";
  }

  // Sla op in Preferences als save=1 in de URL staat
  if (server.hasArg("save") && server.arg("save") == "1") {
    defaultScreen = screenName;
    prefs.putString("def_scr", defaultScreen);

    if (hasCustomCoords) {
      if (screenName == "rog") {
        rogX = moveX;
        rogY = moveY;
        prefs.putInt("rog_x", rogX);
        prefs.putInt("rog_y", rogY);
      } else if (screenName == "custom") {
        customX = moveX;
        customY = moveY;
        prefs.putInt("cust_x", customX);
        prefs.putInt("cust_y", customY);
      } else {
        lgX = moveX;
        lgY = moveY;
        prefs.putInt("lg_x", lgX);
        prefs.putInt("lg_y", lgY);
        //lgY = moveY;  // Let op één kleine bugfix: In het save=1 blok voor het lg-scherm stond hierboven per ongeluk lgY = moveY; in plaats van prefs.putInt("lg_y", lgY);—vervang die regel even door prefs.putInt("lg_y", lgY); zodat ook de Y-waarde van je LG-scherm netjes in het flashgeheugen bewaard blijft.
      }
      activeMode = "Opgeslagen & Default gezet op: " + screenName;
    } else {
      activeMode = "Default scherm gezet op: " + screenName;
    }
  }

  // Voer de muisbeweging en klik uit
  triggerSkipAd(moveX, moveY);

  String response = activeMode + " | X: " + String(moveX) + ", Y: " + String(moveY);
  server.send(200, "text/plain", response);
}

// --- MEDIA ACTIONS ---
void triggerSkipAd(int totalX, int totalY) {
  setLed(150, 0, 150); // Purple flash

  // 1. Reset cursor to top-left corner (LG screen)
  for (int i = 0; i < 40; i++) {
    Mouse.move(-127, -127);
    delay(5);
  }

  // 2. Verdeel het totaal aantal stappen netjes over MOVE_REPEAT (10) bewegingen
  int stepX = totalX / MOVE_REPEAT;
  int remX  = totalX % MOVE_REPEAT;
  int stepY = totalY / MOVE_REPEAT;
  int remY  = totalY % MOVE_REPEAT;

  for (int i = 0; i < MOVE_REPEAT; i++) {
    // Voeg de restwaarde toe aan de eerste beweging(en) voor maximale precisie
    int moveX = stepX + (i < abs(remX) ? (remX > 0 ? 1 : -1) : 0);
    int moveY = stepY + (i < abs(remY) ? (remY > 0 ? 1 : -1) : 0);
    
    Mouse.move(moveX, moveY);
    delay(10);
  }

  delay(30);
  Mouse.click(MOUSE_LEFT);

  delay(200);
  setLed(0, 30, 0);
}

// Overload zonder parameters voor je fysieke knop (BTN_SKIP_PIN)
void triggerSkipAd() {
  if (defaultScreen == "rog") {
    triggerSkipAd(rogX, rogY);
  } else if (defaultScreen == "custom") {
    triggerSkipAd(customX, customY);
  } else {
    triggerSkipAd(lgX, lgY);
  }
}


void triggerPlayPause() {
  Keyboard.write(' ');
  flashLed(0, 150, 150); // Cyan flash
}

void triggerVolUp() {
  ConsumerControl.press(CONSUMER_CONTROL_VOLUME_INCREMENT);
  ConsumerControl.release();
  flashLed(0, 150, 0); // Bright Green flash
}

void triggerVolDown() {
  ConsumerControl.press(CONSUMER_CONTROL_VOLUME_DECREMENT);
  ConsumerControl.release();
  flashLed(150, 150, 0); // Yellow flash
}

void triggerFullscreen() {
  Keyboard.write('f');
  flashLed(0, 80, 200); // Blue flash
}

void triggerSubtitles() {
  Keyboard.write('c');
  flashLed(200, 80, 0); // Orange flash
}

void triggerBack5sec() {
  Keyboard.write(KEY_LEFT_ARROW);
  flashLed(200, 80, 0); // Orange flash
}

void triggerForward5sec() {

  flashLed(200, 80, 0); // Orange flash
}

void triggerKeyDown() {
  Keyboard.write(KEY_DOWN_ARROW);
  flashLed(200, 80, 0); // Orange flash
}

void triggerKeyUp() {
  Keyboard.write(KEY_UP_ARROW);
  flashLed(200, 80, 0); // Orange flash
}

void loadNetworkPrefs() {
  prefs.begin("net_cfg", false);
  staticIP  = prefs.getString("ip", "192.168.5.161");
  staticGW  = prefs.getString("gw", "192.168.5.1");
  staticSN  = prefs.getString("sn", "255.255.255.0");
  staticDNS = prefs.getString("dns", "8.8.8.8");
  prefs.end();
}

void saveNetworkPrefs(IPAddress ip, IPAddress gw, IPAddress sn, IPAddress dns) {
  prefs.begin("net_cfg", false);
  // If user cleared the IP or set 0.0.0.0, store empty strings so DHCP is used
  if (ip == IPAddress(0, 0, 0, 0)) {
    prefs.putString("ip", "");
    prefs.putString("gw", "");
    prefs.putString("sn", "");
    prefs.putString("dns", "");
  } else {
    prefs.putString("ip", ip.toString());
    prefs.putString("gw", gw.toString());
    prefs.putString("sn", sn.toString());
    prefs.putString("dns", dns.toString());
  }
  prefs.end();
}

void startWiFiManager(bool forcePortal) {
  WiFiManager wm;
  shouldSaveConfig = false;

  // Give your NeoPixel a distinct color (Blue) while portal is active
  setLed(0, 0, 180);

  wm.setSaveConfigCallback(saveConfigCallback);
  wm.setConnectTimeout(20);       // Wait 20s for saved Wi-Fi before opening AP
  wm.setConfigPortalTimeout(180); // Auto-close portal after 3 mins if untouched

  // Enable Static IP fields in the portal using saved values (or 0.0.0.0 for DHCP)
  IPAddress _ip, _gw, _sn, _dns;
  _ip.fromString(staticIP.length() > 0 ? staticIP : "192.168.5.161");
  _gw.fromString(staticGW.length() > 0 ? staticGW : "192.168.5.1");
  _sn.fromString(staticSN.length() > 0 ? staticSN : "255.255.255.0");
  _dns.fromString(staticDNS.length() > 0 ? staticDNS : "8.8.8.8");

  wm.setSTAStaticIPConfig(_ip, _gw, _sn, _dns);

  bool connected = false;
  if (forcePortal) {
    if (ERASE_WIFI_ON_HOLD) {
      wm.resetSettings(); // Wipes saved SSID/password
    }
    // Starts AP named "MediaRemote-Setup" (open, or add a second password arg)
    connected = wm.startConfigPortal("MediaRemote-Setup");
  } else {
    connected = wm.autoConnect("MediaRemote-Setup");
  }

  // Save new Static IP / DHCP choice if the user clicked Save in the portal
  // if (shouldSaveConfig) {
  //   saveNetworkPrefs(WiFi.localIP(), WiFi.gatewayIP(), WiFi.subnetMask(), WiFi.dnsIP(0));
  //   loadNetworkPrefs();
  // }

  // 

// Important Tip When Switching to DHCP (0.0.0.0)
// If you type 0.0.0.0 in the portal to switch from a Static IP back to DHCP, WiFi.localIP() will already hold the new router-assigned DHCP address by the time saveConfigCallback() finishes. To make sure your /wifi status page still displays DHCP instead of treating that assigned IP as a static IP on the next reboot, check esp_netif_dhcpc_get_status before saving:
  if (shouldSaveConfig) {
    esp_netif_t* netif = esp_netif_get_handle_from_ifkey("WIFI_STA_DEF");
    esp_netif_dhcp_status_t status;
    if (netif != nullptr && esp_netif_dhcpc_get_status(netif, &status) == ESP_OK && status == ESP_NETIF_DHCP_STARTED) {
      saveNetworkPrefs(IPAddress(0,0,0,0), IPAddress(0,0,0,0), IPAddress(0,0,0,0), IPAddress(0,0,0,0));
    } else {
      saveNetworkPrefs(WiFi.localIP(), WiFi.gatewayIP(), WiFi.subnetMask(), WiFi.dnsIP(0));
    }
    loadNetworkPrefs();
  }

  if (connected) {
    setLed(0, 30, 0); // Dim green = connected
  } else {
    setLed(180, 0, 0); // Red = failed/timed out
    delay(1500);
    ESP.restart();
  }
}

unsigned long wifiBtnPressStart = 0;
bool wifiBtnHandled = false;

void checkWiFiResetButton() {
  if (digitalRead(BTN_WIFI_RESET_PIN) == LOW) {
    if (wifiBtnPressStart == 0) {
      wifiBtnPressStart = millis();
      wifiBtnHandled = false;
    } else if (!wifiBtnHandled && (millis() - wifiBtnPressStart >= 3000)) {
      wifiBtnHandled = true;
      
      // Visual confirmation: 3 quick blue flashes before opening portal
      for (int i = 0; i < 3; i++) {
        setLed(0, 0, 200);
        delay(120);
        setLed(0, 0, 0);
        delay(120);
      }

      server.stop();          // Free port 2222 / network stack cleanly
      startWiFiManager(true); // Open AP portal
      server.begin();         // Restart your remote web server once connected
    }
  } else {
    wifiBtnPressStart = 0;
    wifiBtnHandled = false;
  }
}

  server.on("/pause", []() { triggerPlayPause(); server.send(200, "text/plain", "Action: Play / Pause"); });
  server.on("/back5sec", []() { triggerBack5sec(); server.send(200, "text/plain", "Action: 5 Secs Back"); });
  server.on("/forward5sec", []() { triggerForward5sec(); server.send(200, "text/plain", "Action: 5 Secs Forward"); });

  server.on("/keyup", []() { triggerKeyUp(); server.send(200, "text/plain", "Action: Key Up"); });

  server.on("/wifi", handleWifi);

  //httpUpdater.setup(&server, "/update");

// Futuristic Browser OTA (/update)
  server.on("/update", HTTP_GET, handleUpdatePage);
  server.on("/update", HTTP_POST, []() {
    server.sendHeader("Connection", "close");
    server.send(Update.hasError() ? 500 : 200, "text/plain", Update.hasError() ? "FAIL" : "OK");
    delay(300);
    ESP.restart();
  }, []() {
    HTTPUpload& upload = server.upload();
    if (upload.status == UPLOAD_FILE_START) {
      setLed(180, 100, 0); // Orange LED during flash
      if (!Update.begin(UPDATE_SIZE_UNKNOWN)) {
        Update.printError(Serial);
      }
    } else if (upload.status == UPLOAD_FILE_WRITE) {
      if (Update.write(upload.buf, upload.currentSize) != upload.currentSize) {
        Update.printError(Serial);
      }
    } else if (upload.status == UPLOAD_FILE_END) {
      if (Update.end(true)) {
        setLed(0, 180, 0); // Green on success
      } else {
        setLed(180, 0, 0); // Red on failure
        Update.printError(Serial);
      }
    }
  });
  //server.on("/manual.pdf", handleManual);
  server.on("/manual", handleQuickManual);
  server.on("/manual.pdf", handleManualPdf);


  // 2. Arduino IDE OTA (draadloze poort in Tools > Port)
  ArduinoOTA.setHostname("esp32-mediaremote");
  // Optioneel wachtwoord: ArduinoOTA.setPassword("admin");

  ArduinoOTA.onStart([]() {
    setLed(180, 100, 0); // Oranje LED tijdens firmware update
  });

  ArduinoOTA.onEnd([]() {
    setLed(0, 180, 0);   // Helder groen klaar met flashen
  });

  ArduinoOTA.onError([](ota_error_t error) {
    setLed(180, 0, 0);   // Rood bij fout
  });

  ArduinoOTA.begin();


  server.begin();
  setLed(0, 30, 0); // Dim Green = Ready
}

void loop() {
  ArduinoOTA.handle();
  server.handleClient();
  checkWiFiResetButton();

  if (digitalRead(BTN_SKIP_PIN) == LOW) {
    triggerSkipAd();
    delay(400);
  }

  if (digitalRead(BTN_PAUSE_PIN) == LOW) {
    triggerPlayPause();
    delay(400);
  }

  delay(10);
}
